authRoute.post("/register", async (req, res) => {

        if (!req.body) {
            res.status(400)
            return res.end("Incorrect data")
        } else if (!req.body.login || req.body.login.length < 4) {
            res.status(400)
            return res.end("The login is missing or shorter than 4 characters")
        } else if (!req.body.password || req.body.password.length < 8) {
            res.status(400)
            return res.end("Password is missing or less than 8 characters long")
        } 

        const user = await userService.createUser(req.body.login, req.body.password)
        if (user === null) {
            res.status(400)
            return res.end("user already exist")
        }
        const token = jwt.getToken({ id: user.id, login: user.login })
        res.cookie('accessToken', token, {
            secure: false, 
            sameSite: 'lax',
            path: "/"
        });
        return res.send("Registration was successful.");
    } catch (err) {
        console.log(err)
        res.status(500)
        return res.send("server error")
    }
})

import WebSocket, { WebSocketServer } from 'ws';
import { aiGameSerice, AiGameService } from "../services/game/ai.game.service"
import * as jwt from "./../services/authorization/jwt.service"
import { server } from './../main'
import { PVPGameService, pvpGameService } from '../services/game/pvp.game.service';

function onSocketError(err: any) {
  console.error(err);
}

interface WebSocketData {
  playerId: number | undefined
  ws: WebSocket
}

// export const webSocketMap = new Map<number, WebSocketData>()

export function startWebSocket() {
  const wss = new WebSocketServer({ noServer: true });
  console.log("Вебсокет сервер запущен...")
  wss.on('connection', (ws, req,) => {
    let playerId: number | null = null
    let typeOfGame: "ai" | "pvp" | "qr" | null = null
    let roomId: number | null = null

    const params = new URL(req.url ?? "", `http://${req.headers.host}`).searchParams;

    const token = params.get("token");

    if (token) {
      const payload = jwt.verifyToken(token);
      playerId = payload ? payload.id ?? null : null;
    }

    const gameType = params.get("type");
    if (gameType === "ai" || gameType === "pvp" || gameType === "qr") {
      typeOfGame = gameType;
    }

    const room = params.get("room");
    if (room !== null) {
      const parsedRoomId = Number(room);
      roomId = Number.isInteger(parsedRoomId) ? parsedRoomId : null;
    }

    let service: AiGameService | PVPGameService

    if (gameType === "ai") {
      service = aiGameSerice
      service.createGame(ws, playerId)
    } else if (gameType === "pvp"){
      service = pvpGameService
      service.findGame(ws, playerId)
    }

    ws.on('message', (message) => {
      service.onMessage(message, ws)
    });

    ws.on('close', () => {
      service.onClose(ws)
    });
  });

  server.on('upgrade', function upgrade(request, socket, head) {

    socket.on('error', onSocketError);
    socket.removeListener('error', onSocketError);

    wss.handleUpgrade(request, socket, head, function done(ws) {
      wss.emit('connection', ws, request);
    });
  });

}


===== app/services/authorization/jwt.service.ts =====
import crypto from 'crypto'
import dotenv from 'dotenv';
dotenv.config()

interface PayLoad {
    id: number,
    login: string,
}

export function getToken(payloadBody: PayLoad) {
    if (!process.env.SECRET_KEY) {
        throw new Error("The SECRET_KEY environment variable is not set.")
    }
    const header = { alg: 'HS256', typ: 'JWT' }
    const payload = JSON.stringify(payloadBody)
    let signature = `${encode(header)}.${encode(payload)}`
    const hashedSignature = crypto.createHmac("sha256", process.env.SECRET_KEY).update(signature).digest("base64url")
    return `${encode(header)}.${encode(payload)}.${hashedSignature}`
}

export function verifyToken(token: string) {
    if (!process.env.SECRET_KEY) {
        throw new Error("The SECRET_KEY environment variable is not set.")
    }
    if (typeof token !== "string") {
        return false
    }
    let parts = token.split('.')
    if (parts.length !== 3) {
        return false
    }
    if (crypto.createHmac("sha256", process.env.SECRET_KEY).update(`${parts[0]}.${parts[1]}`).digest("base64url") !== parts[2]) {
        return false
    } else {
        return JSON.parse(decode(parts[1])) as PayLoad
    }
}

export function hashPassword(password: string) {
    return crypto.createHash("sha256").update(password).digest("base64url")
}

function encode(data: any): string {
    let jsonData = JSON.stringify(data)
    const buffer = Buffer.from(jsonData)
    return buffer.toString("base64url")
}

function decode(string: string): string {
    let buffer = Buffer.from(string, "base64url")
    return JSON.parse(buffer.toString())
}

import WebSocket from "ws";

export interface Card {

    name: string;
    suit: string;
    value: number;
}

interface SendMessage {
    enemyEvent?: Event,
    win?: "none" | "user" | "enemy",
    winMessage?: string,
    userDeck?: Card[],
    takeCard?: Card,
    countOfDeckEnemy?: number,

type WhoseTurnEnum = 'user' | 'enemy'

export type Event = "take" | "pass"

export default class AiGameRoom {
    ws: WebSocket
    playerId: number | null
    hasStart = false
    whoseTurn: WhoseTurnEnum = "user"
    stepCount = 0
    historyOfGame: string[] = []
    cardsOfUser: Card[] = []
    cardsOfEnemy: Card[] = []
    userIsPass = false
    enemyIsPass = false
    mainDeck: Card[] = []
    raitingOfPlayerBeforeGame: number = 0
    gameId: number = 0

    constructor(ws: WebSocket, playerId: number | null) {
        this.mainDeck = deck
        this.ws = ws
        this.playerId = playerId
    }

    get valueOfEnemyDeck() {
        return this.cardsOfEnemy.reduce((prev, curr) => { return prev += curr.value }, 0)
    }
    get valueOfUserDeck() {
        return this.cardsOfUser.reduce((prev, curr) => { return prev += curr.value }, 0)
    }

    closeGame() {
        gameService.abandonedGame(this.gameId)
    }

    resetGame() {
        this.hasStart = false
        this.whoseTurn = "user"
        this.stepCount = 0
        this.historyOfGame = []
        this.cardsOfUser = []
        this.cardsOfEnemy = []
        this.userIsPass = false
        this.enemyIsPass = false
        this.mainDeck = []
        this.mainDeck = deck.slice()
    }

    async startGame() {
        if (!this.hasStart) {
            if (this.playerId) {
                const data = await userService.getRatingById(this.playerId, "ai")
                this.raitingOfPlayerBeforeGame = data ? data : 0
                if (this.playerId)
                    this.gameId = await gameService.createGame(this.playerId, 0, "ai", this.raitingOfPlayerBeforeGame, 0)
            }
            this.hasStart = true
            let card = this.enemyTakeCard();
            this.setHistoryOfGame("Начало игры");

            card = this.userTakeCard();
            this.send({
                takeCard: card,
                whoseTurn: "user",
                userDeck: this.cardsOfUser,
                countOfDeckEnemy: this.cardsOfEnemy.length
            })

        }
    }

    enemyTakeCard() {
        let index = randomaizer(0, this.mainDeck.length - 1);
        const card = this.mainDeck[index]
        this.addCardToEnemyDeck(this.mainDeck[index]);
        this.mainDeck.splice(index, 1)
        if (this.playerId) {
            gameService.madeStep(this.gameId, 0, "take", this.stepCount, this.valueOfEnemyDeck, card.name + " " + card.suit)
        }
        this.stepCount += 1
        return card
    }
    //КОСЯК
    userTakeCard() {
        let index = randomaizer(0, this.mainDeck.length - 1);
        const card = this.mainDeck[index]
        this.addCardToUserDeck(this.mainDeck[index]);
        this.mainDeck.splice(index, 1)
        if (this.playerId) {
            gameService.madeStep(this.gameId, this.playerId, "take", this.stepCount, this.valueOfUserDeck, card.name + " " + card.suit)
        }
        this.stepCount += 1
        return card

    setUserIsPass() {

    setEnemyIsPass() {

        this.cardsOfUser.push(card)

    addCardToEnemyDeck(card: Card) {
        this.cardsOfEnemy.push(card)
    }
    incrementStepCount() {
        this.stepCount += 1
    }
    setWhoseTurn(who: WhoseTurnEnum) {
        this.whoseTurn = who
    }

    setHistoryOfGame(event: string) {
        this.historyOfGame.push(event)
    }

    async win(who: "user" | "enemy" | "none", message: string) {
        const finalRes = await getAction(this.historyOfGame, this.valueOfEnemyDeck)
        this.ws.send(JSON.stringify({
            enemyMessage: finalRes.message,

    }

    async madeStep(event: Event): Promise<void> {
        console.log(this.playerId)
        this.setWhoseTurn("enemy")
        this.ws.send(JSON.stringify({ whoseTurn: "enemy" }))
        if (event == "take") {
            const card = this.userTakeCard()
            this.send({ takeCard: card, userDeck: this.cardsOfUser, })
            this.setHistoryOfGame("Пользователь взял карту")

            } else {
                if (this.enemyIsPass) {
                    if (!this.enemyIsPass && this.playerId) {
                        gameService.madeStep(this.gameId, 0, "pass", this.stepCount, this.valueOfEnemyDeck)
                        this.stepCount++
                    }
                    this.enemyIsPass = true
                    this.send({ takeCard: card, userDeck: this.cardsOfUser, countOfDeckEnemy: this.cardsOfEnemy.length })
                } else {
                    const res = await getAction(this.historyOfGame, this.valueOfEnemyDeck)
                    if (res.event == "take" && !this.enemyIsPass) {
                        const card = this.enemyTakeCard()
                        this.send({ enemyMessage: res.message })
                        if (this.valueOfEnemyDeck > 21) {
                            this.setHistoryOfGame("Противник набрал больше 21-го очка. Игрок выйграл")
                            this.win("user", "Противник набрал больше 21-го очка. Вы выйграли!! :)")
                        } else {
                            this.setHistoryOfGame(`Противник получает карту ${card.suit} ${card.name}`)
                            this.setWhoseTurn('user')
                            this.send({ enemyEvent: "take", countOfDeckEnemy: this.cardsOfEnemy.length, whoseTurn: 'user' })
                        }

                        this.send({ enemyEvent: "pass", enemyMessage: res.message, whoseTurn: "user" })
                        this.setEnemyIsPass()
                        this.setWhoseTurn("user")
                    }
                }

import { Event } from "./AiGameRoom"
import WebSocket from "ws";

export const gameRoomMap = new Map<WebSocket, GameRoom>()

export class AiGameService {
    static instance: AiGameService
    gameRoomMap = new Map<WebSocket, GameRoom>()

    constructor(){
        if(AiGameService.instance){
            return AiGameService.instance
        }
        AiGameService.instance = this
    }

    onConnect(ws: WebSocket, player1_Id: number | null){
        ws.send(JSON.stringify({ whoseTurn: "start" }))
    }

    onMessage(message: WebSocket.RawData, ws: WebSocket) {
        try {
            const parsedMessage = JSON.parse(String(message))
            if (!parsedMessage.event) {
                throw Error("invalidate data")
            }
            if (parsedMessage.event === "start") {
                this.startGame(ws)
                console.log("Партия начата")
            } else {
                this.madeStep(ws, parsedMessage.event)
                console.log("Сделан ход")
            }
        } catch (err) {
            console.log(err)
        }
    }

    onClose(ws: WebSocket){
        this.deleteRoom(ws)
    }

    createGame(ws: WebSocket, playerId: number | null) {
        if (!this.gameRoomMap.has(ws)) {
            this.gameRoomMap.set(ws, new GameRoom(ws, playerId))
        }
        const room = gameRoomMap.get(ws)
        if (room) {
            room.resetGame()
            ws.send(JSON.stringify({ info: "game is created" }))
        } else {
            new Error("Game room not found")
        }
        console.log("Партия создана в базе данных")
        return room
    }

    startGame(ws: WebSocket) {
        const room = this.gameRoomMap.get(ws)
        if (room) {
            void room.startGame().catch((error) => {
                console.error("Failed to start AI game:", error)
            })
        } else {
            new Error("Game room not found")
        }
        return room
    }

    madeStep(ws: WebSocket, event: Event) {
        const gameRoom = this.gameRoomMap.get(ws)
        if (!gameRoom) {
            new Error("Game room not found")
        }
        gameRoom?.madeStep(event)
    }

    deleteRoom(ws: WebSocket) {
        const gameRoom = this.gameRoomMap.get(ws)
        if (!gameRoom) {
            new Error("Game room not found")
        }
        gameRoom?.closeGame()
        this.gameRoomMap.delete(ws)
    }
}

import { DeepPartial } from "typeorm"
import AppDataSource from "../../AppDataSource"
import { ACTION, GameMoves } from "../../entity/GameMoves"
import { Games, GameType, STATUS } from "../../entity/Games"
import { Users } from "../../entity/Users"

export async function createGame(player1_id: number, player2_id: number, type: "ai" | "pvp", player1_score: number, player2_score: number) {
    const repository = AppDataSource.getRepository(Games)
    const game = repository.create({
        player1_id: player1_id !== 0 ? { id: player1_id } : undefined,
        player2_id: player2_id !== 0 ? { id: player2_id } : undefined,
        game_type: type === "ai" ? GameType.AI : GameType.PVP,
        player1_score: player1_score,
        player2_score: player2_score,

    })
    return (await repository.save(game)).id

}

export async function finishGame(whoWin_id: number, game_id: number) {
    const repository = AppDataSource.getRepository(Games)
    const game = await repository.findOneBy({ id: game_id })
    if (game) {
        game.status = STATUS.FINISHED
        if (whoWin_id !== 0) {
            game.winner_id = { id: whoWin_id } as Users
        }
        game.ended_at = new Date()
        repository.save(game)
    }
}

export async function abandonedGame(game_id: number) {
    const repository = AppDataSource.getRepository(Games)
    const game = await repository.findOneBy({ id: game_id })
    if (game && game.status !== STATUS.FINISHED) {
        game.status = STATUS.ABANDONED
        game.ended_at = new Date()
        repository.save(game)
    }
}

export async function madeStep(game_id: number, player_id: number, event: "take" | "pass", stepCount: number, score: number, card?: string) {
    const repository = AppDataSource.getRepository(GameMoves)
    const move = await repository.create({
        game_id: { id: game_id },
        action: event == "take" ? ACTION.TAKE : ACTION.PASS,
        point_after_move: score,
        step_number: stepCount,
        player_id: player_id !== 0 ? { id: player_id } : undefined,
        card_drawn: event === "take" ? card : undefined
    })
    repository.save(move)
}

                this.stepCount++
            }
            this.player2IsPass = true
            this.setHistoryOfGame("Игрок 2 пасс")
            if (this.player1IsPass) {
                if (this.valueOfPlayer2Deck > this.cardsOfPlayer1.reduce((prev, curr) => { return prev += curr.value }, 0)) {
                    this.setHistoryOfGame("Игрок 2 набрал больше очков чем игрок 1. Игрок 2 выйграл")
                    this.win("player2")
                    this.send({ win: "user", winMessage: "Вы набрали больше очков чем противник. Вы выйграли :)" }, "player2")
                    this.send({ win: "enemy", winMessage: "Противник набрал больше очко чем вы. Вы проиграли :(" }, "player1")

import { Event } from "./PVPGameRoom"
import WebSocket from "ws";
import PVPGameRoom from "./PVPGameRoom";



interface PlayerInQueue {
    ws: WebSocket,
    player_id?: number
}

export class PVPGameService {
    static instance: PVPGameService
    playerQueue: PlayerInQueue[] = []
    gameRoomMap = new Map<WebSocket[], PVPGameRoom>()

    constructor() {
        if (PVPGameService.instance) {
            return PVPGameService.instance
        }
        PVPGameService.instance = this
    }

    onConnect(ws: WebSocket, player_Id: number | null) {
        ws.send(JSON.stringify({ whoseTurn: "start", info: "finding enemy" }))
        this.findGame(ws, player_Id)
        console.log("player is connected")
    }

    onMessage(message: WebSocket.RawData, ws: WebSocket) {
        try {
            const parsedMessage = JSON.parse(String(message))
            if (!parsedMessage.event) {
                throw Error("invalidate data")
            }
            if (parsedMessage.event === "start") {
                let gameRoom: PVPGameRoom | null = null
                for (let arr of this.gameRoomMap) {
                    if (arr[0].includes(ws)) {
                        gameRoom = arr[1]
                    }
                }
                gameRoom?.playerIsReady(ws)
            } else {
                this.madeStep(ws, parsedMessage.event)
                console.log("Сделан ход")
            }


        } catch (err) {
            console.log(err)
        }
    }

    onClose(ws: WebSocket) {
        this.playerQueue.forEach((value, key) => {
            if (value.ws === ws){
                console.log(this.playerQueue)
                this.playerQueue.splice(key, 1)
                console.log(this.playerQueue)
                return
            }
        })
        
        this.deleteRoom(ws)
    }

    findGame(ws: WebSocket, player_id: number | null) {
        if (this.playerQueue.length === 0) {
            this.playerQueue.push({ ws: ws, player_id: player_id ? player_id : undefined })
            console.log("Добавили в очередь")
            console.log(this.playerQueue)
        } else {
            const player2 = this.playerQueue.shift()
            if (player2) {
                this.gameRoomMap.set([ws, player2.ws], new PVPGameRoom(ws, player_id, player2.ws, player2.player_id ?? null))
                console.log(this.playerQueue)
                console.log("Очередь после того как нашли пользователя")
                console.log(ws === player2.ws)
                ws.send(JSON.stringify({ info: "enemy is founded", whoseTurn: "start" }))
                player2.ws.send(JSON.stringify({ info: "enemy is founded", whoseTurn: "start" }))
            } else {
                console.log("Пользователь не найден в очереди pvp.game.service")
            }
        }
    }

    madeStep(ws: WebSocket, event: Event) {
        let gameRoom: PVPGameRoom | null = null
        for (let arr of this.gameRoomMap) {
            if (arr[0].includes(ws)) {
                gameRoom = arr[1]
            }
        }
        if (!gameRoom) {
            new Error("Game room not found")
        }
        gameRoom?.madeStep(event, ws)
    }

    deleteRoom(ws: WebSocket) {
        let gameRoom: PVPGameRoom | null = null
        let key: WebSocket[] | null = null
        for (let arr of this.gameRoomMap) {
            if (arr[0].includes(ws)) {
                gameRoom = arr[1]
                key = arr[0]
            }
        }
        if (!gameRoom || !key) {
            new Error("Game room not found")
        } else {
            gameRoom?.closeGame()
            this.gameRoomMap.delete(key)
        }
    }
}

export const pvpGameService = new PVPGameService()

            console.log(resultObject)
            throw new Error("The AI ​​generated incorrect data.")
        }
        return resultObject;

    }

}

export async function getRatingById(id: number, typeOfRating: "ai" | "pvp") {
    const resp = await AppDataSource.getRepository(UserStats).findOne({
        where: { id: id },
    })
    if (resp && typeOfRating == "ai") {
        return resp.rating_ai
    } else if (resp && typeOfRating == "pvp")
        return resp?.rating_pvp
    else {
        return null
    }
}

export async function getTop10(typeOfRating: "ai" | "pvp") {
    if (typeOfRating === "ai") {
        const resp = await AppDataSource.getRepository(UserStats).find({

            },
            select: {
                id: true,
                rating_ai: true,
                users: {
                    login: true
                }
            },
            take: 10
        })
        return resp
    } else {
        const resp = await AppDataSource.getRepository(UserStats).find({
            order: {
                rating_pvp: 'DESC'
            },
            select: {
                id: true,
                rating_ai: true,
                users: {

export async function createUser(login: string, password: string) {
    try {
        return await AppDataSource.transaction(async (manager) => {
            const user = manager.create(Users, {
                login: login,
                password: jwt.hashPassword(password)
            })
            await manager.save(user)

            const user_stats = manager.create(UserStats, {
                id: user.id,
            })
            await manager.save(user_stats)
            return user
        })
    
    } catch (err: any) {
        if (err.code === '23505') {
            return null
        } else {
            throw err
        }
    }
}

        return resp
    } catch (err) {
        return null
    }
}
